# The market-data server and its ingest (plans/server.md). Included from the top-level file. Both run on the application's own Store
# (src/model): what they keep is kept by the model's stores, in SQLite files or, where libmariadb was found, in MariaDB (src/model/MariaDbStores.h).
# Libraries, all pinned and open source:
#   cpp-httplib  the server's HTTP (MIT, one header)
#   miniz        reads the zip files a dataset downloads (MIT)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
# gzip for the answers (a client sends Accept-Encoding: gzip): only where zlib is found, and it is not required.
find_package(ZLIB QUIET)
if(ZLIB_FOUND)
  set(HTTPLIB_USE_ZLIB_IF_AVAILABLE ON CACHE BOOL "" FORCE)
else()
  set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
endif()
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZSTD_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_COMPILE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(httplib
  GIT_REPOSITORY https://github.com/yhirose/cpp-httplib
  GIT_TAG v0.18.3)
FetchContent_MakeAvailable(httplib)

# miniz is built here from its sources, not as its own CMake project: that one sets CMAKE_BUILD_TYPE to Release in the cache when it is empty,
# which would turn the optimisation on for the whole program. Its export header is generated by that project, and is empty for a static library.
FetchContent_Declare(miniz
  GIT_REPOSITORY https://github.com/richgel999/miniz
  GIT_TAG 3.0.2
  SOURCE_SUBDIR no_cmake_project_here)
FetchContent_MakeAvailable(miniz)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/miniz_include)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/miniz_include/miniz_export.h "#pragma once\n#define MINIZ_EXPORT\n#define MINIZ_NO_EXPORT\n")
add_library(miniz STATIC ${miniz_SOURCE_DIR}/miniz.c ${miniz_SOURCE_DIR}/miniz_tdef.c ${miniz_SOURCE_DIR}/miniz_tinfl.c ${miniz_SOURCE_DIR}/miniz_zip.c)
target_include_directories(miniz SYSTEM PUBLIC ${miniz_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/miniz_include)
set_target_properties(miniz PROPERTIES C_STANDARD 99)
# Only zips in memory are read and written (Zip.cpp), so miniz's file I/O is left out (its fopen path also prints a #pragma message at every build).
target_compile_definitions(miniz PUBLIC MINIZ_NO_STDIO)

# Everything provider-free: the bars, the raw file cache, the ingest and the API.
add_library(market_lib STATIC Bars.cpp Zip.cpp RawCache.cpp Backends.cpp Config.cpp Ingest.cpp Api.cpp)
target_include_directories(market_lib PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_compile_features(market_lib PUBLIC cxx_std_20)
target_link_libraries(market_lib PUBLIC stocks_model stocks_nodes stocks_http miniz nlohmann_json::nlohmann_json httplib::httplib)
# A request line of 100 edge ids (Api.h: the cap) is up to about 11.5 KB with every "/", "," and "#" of the longest ids sent as %XX, over the
# library's 8 KB; the whole line is allowed 16 KB. Every file that includes httplib.h links this library, so they all see the same value.
target_compile_definitions(market_lib PUBLIC CPPHTTPLIB_REQUEST_URI_MAX_LENGTH=16384)
if(TARGET stocks_mariadb)
  target_link_libraries(market_lib PUBLIC stocks_mariadb)
  target_compile_definitions(market_lib PUBLIC MARKET_WITH_MARIADB)
endif()

# The offline datasets (market/offline/): loaded when market-ingest is asked, apart from the sources of src/source/ that run beside the
# browser. The only code in these targets that knows a provider.
add_library(market_offline STATIC offline/HistData.cpp offline/Lme.cpp)
target_link_libraries(market_offline PUBLIC market_lib)

add_executable(market-ingest market_ingest_main.cpp)
target_link_libraries(market-ingest PRIVATE market_lib market_offline)
add_executable(market-server market_server_main.cpp)
target_link_libraries(market-server PRIVATE market_lib)
set_target_properties(market-ingest market-server PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})

# The tests: the keys, the aggregation, the parser, the shift, the ingest with a fake site and the API, on SQLite files, and on MariaDB too
# where MARKET_TEST_DB_USER is set (see tests/MarketTestSupport.h).
set(MARKET_TESTS ${PROJECT_SOURCE_DIR}/tests)
add_executable(market_tests ${MARKET_TESTS}/test_main.cpp ${MARKET_TESTS}/MarketBarsTests.cpp ${MARKET_TESTS}/MarketZipTests.cpp
  ${MARKET_TESTS}/HistDataTests.cpp ${MARKET_TESTS}/LmeTests.cpp ${MARKET_TESTS}/MarketIngestTests.cpp ${MARKET_TESTS}/MarketApiTests.cpp ${MARKET_TESTS}/MarketConfigTests.cpp)
target_link_libraries(market_tests PRIVATE market_lib market_offline)
add_test(NAME market_tests COMMAND market_tests)
set_tests_properties(market_tests PROPERTIES TIMEOUT 120)

# The server reads its port strictly: the whole text is a number from 0 to 65535 (it would otherwise listen somewhere else, or nowhere).
add_test(NAME market_server_rejects_a_port_that_is_not_one COMMAND market-server --port 80x)
set_tests_properties(market_server_rejects_a_port_that_is_not_one PROPERTIES PASS_REGULAR_EXPRESSION "--port is a number from 0 to 65535")
add_test(NAME market_server_rejects_a_port_out_of_range COMMAND market-server --port 65536)
set_tests_properties(market_server_rejects_a_port_out_of_range PROPERTIES PASS_REGULAR_EXPRESSION "--port is a number from 0 to 65535")
